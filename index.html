Demo · model excerptcode on GitHub
# dcf.py — enterprise value from free cash flows
def enterprise_value(fcf, wacc, g):
    pv = sum(cf / (1 + wacc)**t
             for t, cf in enumerate(fcf, 1))
    tv = fcf[-1] * (1 + g) / (wacc - g)
    return pv + tv / (1 + wacc)**len(fcf)
